Drone Config Analyzer

ใส่ข้อมูลสเปคโดรนของคุณ แล้วระบบจะวิเคราะห์ PID, Filter, Rates ให้ทันที พร้อมคำสั่ง CLI ที่ copy & paste เข้า Betaflight ได้เลย

  • วิเคราะห์ PID ตามสเปคมอเตอร์, prop, frame
  • แนะนำ Filter settings อัตโนมัติ
  • สร้าง Betaflight CLI พร้อมใช้งาน
  • ตรวจสอบความเข้ากันได้ของ hardware
ใช้งานเลย
configdoctor_cli_output.sh
# Generated by ConfigDoctor
# 2306 2400KV · 5" props · 4S

set p_pitch = 52
set i_pitch = 90
set d_pitch = 40
set f_pitch = 0
set p_roll = 48
set i_roll = 85
set d_roll = 38
set f_roll = 0

set dyn_lpf_dterm_min_hz = 120
set dyn_lpf_dterm_max_hz = 255
set dterm_lowpass_type = PT1

set gyro_rpm_notch_q = 500
set gyro_rpm_notch_min = 120
save
